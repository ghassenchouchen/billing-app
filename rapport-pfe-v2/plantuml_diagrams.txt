%% 8. DIAGRAMME DE SEQUENCE — CREER UN CLIENT
%% --------------------------------------------

@startuml
actor "Agent commercial" as Agent
participant "API Gateway" as GW
participant "Customer-service" as CS
database "MySQL" as DB

Agent -> GW : POST /api/customers
GW -> CS : POST /customers
CS -> CS : Valider les données\n(CIN 8 chiffres, tél 8 chiffres)
CS -> DB : INSERT Customer
DB --> CS : Customer sauvegardé
CS --> GW : 201 Created (CustomerDto)
GW --> Agent : 201 Created
@enduml


%% 9. DIAGRAMME DE SEQUENCE — SOUSCRIRE UN ABONNEMENT
%% ----------------------------------------------------

@startuml
actor "Agent commercial" as Agent
participant "API Gateway" as GW
participant "Subscription-service" as SS
participant "Customer-service" as CS
participant "Catalog-service" as CAT
database "MySQL" as DB

Agent -> GW : POST /api/abonnements
GW -> SS : POST /abonnements
SS -> CS : GET /customers/{ref}\n(Feign)
CS --> SS : CustomerDto
SS -> CAT : GET /offres/{id}\n(Feign)
CAT --> SS : OffreDto
SS -> DB : INSERT Abonnement
DB --> SS : Abonnement sauvegardé
SS --> GW : 201 Created
GW --> Agent : 201 Created
@enduml


%% 10. DIAGRAMME DE SEQUENCE — GENERER UNE FACTURE
%% -------------------------------------------------

@startuml
actor "Agent commercial" as Agent
participant "API Gateway" as GW
participant "Billing-service" as BS
participant "Subscription-service" as SS
participant "Customer-service" as CS
database "MySQL" as DB

Agent -> GW : POST /api/invoices/generate/{aboId}
GW -> BS : POST /invoices/generate/{aboId}
BS -> SS : GET /abonnements/{id}\n(Feign)
SS --> BS : AbonnementDto
BS -> CS : GET /customers/{ref}\n(Feign)
CS --> BS : CustomerDto
BS -> BS : Calculer montant HT\n+ TVA 19%% = TTC
BS -> DB : INSERT Facture + LigneFacture
DB --> BS : Facture sauvegardée
BS --> GW : 201 Created (FactureDto)
GW --> Agent : 201 Created
@enduml
